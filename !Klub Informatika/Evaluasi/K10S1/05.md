**Soal 25.** Perhatikan kode berikut.

```cpp
#include <iostream>
using namespace std;

int binser(int arr[], int low, int high, int target) {
	if (low <= high) {
		int mid = low + (high - low) / 2;
		
		if (arr[mid] == target)
			return mid;
		if (arr[mid] > target)
			return binser(arr, low, mid - 1, target);
			
		return binser(arr, mid + 1, high, target);
	}
	
	return -1;
}
```
